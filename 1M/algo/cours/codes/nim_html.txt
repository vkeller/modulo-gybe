html_code = """
<div id="nim-game" style="font-family: Arial, sans-serif; background-color: #2c3e50; color: #ecf0f1; padding: 20px; border-radius: 10px; max-width: 600px; margin: auto;">
    <h2 style="text-align: center; color: #ecf0f1; margin-top: 0;">Jeu de Nim (Variante Misère)</h2>
    <p style="text-align: center; font-size: 0.9em; color: #bdc3c7;">Règle : Celui qui prend le dernier objet a <strong>perdu</strong>.</p>
    
    <!-- Zone Nim-somme -->
    <div style="background-color: #34495e; padding: 10px; border-radius: 5px; font-family: monospace; font-size: 0.95em; margin-bottom: 15px; border-left: 4px solid #f1c40f;">
        <div id="nim-details">Configuration : ...</div>
        <div id="nim-somme" style="color: #f1c40f; font-weight: bold;">Nim-somme : ...</div>
    </div>

    <!-- Message de statut / Victoire / Défaite -->
    <div id="status-msg" style="text-align: center; font-weight: bold; padding: 8px; margin-bottom: 15px; border-radius: 4px; display: none;"></div>

    <!-- Zone des tas -->
    <div id="tas-container"></div>

    <!-- Controls -->
    <div style="text-align: center; margin-top: 20px;">
        <button onclick="resetGame()" style="background-color: #27ae60; color: white; border: none; padding: 8px 16px; font-weight: bold; border-radius: 4px; cursor: pointer;">Nouvelle Partie</button>
    </div>
</div>

<script>
(function() {
    let tasInitiaux = [1, 3, 5, 7];
    let tas = [...tasInitiaux];

    function calculerNimSomme() {
        return tas.reduce((acc, val) => acc ^ val, 0);
    }

    function rendreAffichage() {
        const container = document.getElementById('tas-container');
        container.innerHTML = '';

        let ns = calculerNimSomme();
        
        // MàJ texte Nim-Somme
        let details = tas.map((v, i) => `T${i+1}:${v} (${v.toString(2).padStart(3, '0')})`).join('  ');
        document.getElementById('nim-details').innerText = 'Configuration : ' + details;
        document.getElementById('nim-somme').innerText = `Nim-somme     : ${ns} (${ns.toString(2).padStart(3, '0')} [binaire])`;

        // Génération des lignes de tas
        tas.forEach((nbObjets, idx) => {
            let row = document.createElement('div');
            row.style.cssText = 'background-color: #34495e; margin: 8px 0; padding: 8px 12px; border-radius: 5px; display: flex; align-items: center; justify-content: space-between;';

            let info = document.createElement('div');
            info.style.cssText = 'display: flex; align-items: center; gap: 15px;';
            
            let label = document.createElement('span');
            label.style.fontWeight = 'bold';
            label.innerText = `Tas ${idx+1} (${nbObjets})`;

            let visuel = document.createElement('span');
            visuel.style.cssText = 'color: #f39c12; font-weight: bold; font-size: 1.2em; letter-spacing: 4px;';
            visuel.innerText = 'O'.repeat(nbObjets);

            info.appendChild(label);
            info.appendChild(visuel);
            row.appendChild(info);

            let btnGroup = document.createElement('div');
            if (nbObjets === 0) {
                let vide = document.createElement('span');
                vide.style.cssText = 'color: #7f8c8d; font-size: 0.9em; italic;';
                vide.innerText = 'Vide';
                btnGroup.appendChild(vide);
            } else {
                for (let nb = 1; nb <= nbObjets; nb++) {
                    let btn = document.createElement('button');
                    btn.innerText = `-${nb}`;
                    btn.style.cssText = 'background-color: #e67e22; color: white; border: none; padding: 4px 8px; margin-left: 4px; border-radius: 3px; font-weight: bold; cursor: pointer;';
                    btn.onclick = () => jouerCoupJoueur(idx, nb);
                    btnGroup.appendChild(btn);
                }
            }

            row.appendChild(btnGroup);
            container.appendChild(row);
        });
    }

    window.jouerCoupJoueur = function(tasIdx, count) {
        tas[tasIdx] -= count;
        rendreAffichage();

        if (verifierFin()) {
            afficherStatut("Vous avez pris le dernier objet... Vous avez perdu !", "#e74c3c");
            return;
        }

        setTimeout(jouerCoupIA, 600);
    };

    function jouerCoupIA() {
        if (verifierFin()) return;

        let tasSup1 = tas.map((v, i) => v > 1 ? i : -1).filter(i => i !== -1);

        if (tasSup1.length > 0) {
            let ns = calculerNimSomme();
            let coupTrouve = false;

            if (ns !== 0) {
                for (let i = 0; i < tas.length; i++) {
                    let val = tas[i];
                    let cible = val ^ ns;
                    if (cible < val) {
                        let resteApres = tas.map((v, idx) => idx === i ? cible : v);
                        if (Math.max(...resteApres) <= 1) {
                            let count1 = resteApres.filter(v => v === 1).length;
                            if (count1 % 2 === 0) {
                                cible = (count1 + 1) % 2 !== 0 ? 1 : 0;
                                if (cible >= val) cible = 0;
                            }
                        }
                        tas[i] = cible;
                        coupTrouve = true;
                        break;
                    }
                }
            }

            if (!coupTrouve) {
                tas[tasSup1[0]] -= 1;
            }
        } else {
            let tasDe1 = tas.map((v, i) => v === 1 ? i : -1).filter(i => i !== -1);
            tas[tasDe1[0]] = 0;
        }

        rendreAffichage();

        if (verifierFin()) {
            afficherStatut("L'ordinateur a été forcé de prendre le dernier objet. Vous avez gagné !", "#27ae60");
        }
    }

    function verifierFin() {
        return tas.reduce((a, b) => a + b, 0) === 0;
    }

    function afficherStatut(msg, bgCouleur) {
        let status = document.getElementById('status-msg');
        status.style.display = 'block';
        status.style.backgroundColor = bgCouleur;
        status.innerText = msg;
    }

    window.resetGame = function() {
        tas = [...tasInitiaux];
        document.getElementById('status-msg').style.display = 'none';
        rendreAffichage();
    };

    // Premier rendu au chargement de la cellule
    rendreAffichage();
})();
</script>
